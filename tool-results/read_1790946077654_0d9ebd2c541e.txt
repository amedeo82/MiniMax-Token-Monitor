    45→const SETTINGS_KEY = "mm-dash-settings-v1";

    47→const HISTORY_MAX = 600;
    48→
    49→function loadSettings(): DashboardSettings {
    50→  try {
    51→    const raw = localStorage.getItem(SETTINGS_KEY);
    52→    if (!raw) return DEFAULT_SETTINGS;
    53→    const parsed = JSON.parse(raw) as Partial<DashboardSettings>;
    54→    return { ...DEFAULT_SETTINGS, ...parsed };
    55→  } catch {
    56→    return DEFAULT_SETTINGS;
    57→  }
    58→}
    59→
    60→function loadHistory(): HistoryPoint[] {
    61→  try {
    62→    const raw = localStorage.getItem(HISTORY_KEY);
    63→    if (!raw) return [];
    64→    const parsed = JSON.parse(raw) as HistoryPoint[];
    65→    return Array.isArray(parsed) ? parsed.slice(-HISTORY_MAX) : [];
    66→  } catch {
    67→    return [];
    68→  }
    69→}

   310→    try {
   311→      localStorage.setItem(SETTINGS_KEY, JSON.stringify(s));

   313→      /* noop */

   316→    setError(null);
   317→    void fetchData(s);

   325→        region: settings.region,

   336→      setError(null);
   337→      pushHistory(snap);

   340→      const message = e instanceof Error ? e.message : "JSON non interpretabile.";
   341→      setError({ code: "parse", message });
   342→      toast.error(message);
   343→    }
   344→  };

   354→  const otherModels = snapshot ? snapshot.models.filter((m) => m.modelName.toLowerCase() !== "general") : [];

   376→  const selectedWindowEnd = selectedWindow?.end ?? null;
   377→  const selectedWindowRemaining = selectedWindow?.remainingPercent ?? null;
   378→

   380→  return (
   381→    <div className="relative min-h-screen bg-[#09090b] text-zinc-100">
   382→      {/* Decorazioni di sfondo */}
   383→      <div
   384→        aria-hidden
   385→        className="pointer-events-none fixed inset-0"
   386→        style={{

   389→        }}

   392→      <div className="relative flex min-h-screen flex-col">
   393→        {/* ------------------------------ Header ------------------------------ */}

   395→          <div className="mx-auto flex max-w-7xl flex-wrap items-center gap-3 px-4 py-3 sm:px-6">
   396→            <div className="flex items-center gap-3">
   397→              <div className="flex size-10 items-center justify-center rounded-xl bg-gradient-to-br from-rose-500 to-orange-500 shadow-lg shadow-rose-500/20">
   398→                <Zap className="size-5 text-white" />

   409→                <MapPin className="size-3.5 text-rose-400" />
   410→                <div className="text-right leading-tight">
   411→                  <p className="font-mono text-sm font-semibold tabular-nums text-zinc-100">{mounted ? nowRomeTime() : "--:--:--"}</p>
