    60→const SETTINGS_KEY = "mm-dash-settings-v1";

    63→
    64→function loadSettings(): DashboardSettings {
    65→  try {
    66→    const raw = localStorage.getItem(SETTINGS_KEY);
    67→    if (!raw) return DEFAULT_SETTINGS;
    68→    const parsed = JSON.parse(raw) as Partial<DashboardSettings>;
    69→    return { ...DEFAULT_SETTINGS, ...parsed };
    70→  } catch {
    71→    return DEFAULT_SETTINGS;
    72→  }
    73→}
    74→
    75→function loadHistory(): HistoryPoint[] {
    76→  try {
    77→    const raw = localStorage.getItem(HISTORY_KEY);
    78→    if (!raw) return [];
    79→    const parsed = JSON.parse(raw) as HistoryPoint[];
    80→    return Array.isArray(parsed) ? parsed.slice(-HISTORY_MAX) : [];
    81→  } catch {
    82→    return [];
    83→  }
    84→}

   460→  const selectedWindow = chartWindow === "interval" ? general?.interval : general?.weekly;
   461→  const selectedWindowEnd = selectedWindow?.end ?? null;
   462→  const selectedWindowRemaining = selectedWindow?.remainingPercent ?? null;
   463→

   467→      {/* Decorazioni di sfondo */}
   468→      <div
   469→        aria-hidden
   470→        className="pointer-events-none fixed inset-0"
   471→        style={{
   472→          background:
   473→            "radial-gradient(700px 380px at 12% -5%, rgba(244,63,94,0.10), transparent 60%), radial-gradient(800px 420px at 88% 0%, rgba(245,158,11,0.07), transparent 60%)",
   474→        }}
   475→      />
   476→
   477→      <div className="relative flex min-h-screen flex-col">
   478→        {/* ------------------------------ Header ------------------------------ */}
