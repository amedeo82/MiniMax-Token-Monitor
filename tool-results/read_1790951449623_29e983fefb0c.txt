    56→function loadSettings(): DashboardSettings {
    57→  try {
    58→    const raw = localStorage.getItem(SETTINGS_KEY);
    59→    if (!raw) return DEFAULT_SETTINGS;
    60→    const parsed = JSON.parse(raw) as Partial<DashboardSettings>;
    61→    return { ...DEFAULT_SETTINGS, ...parsed };
    62→  } catch {
    63→    return DEFAULT_SETTINGS;
    64→  }
    65→}
    66→
    67→function loadHistory(): HistoryPoint[] {
    68→  try {
    69→    const raw = localStorage.getItem(HISTORY_KEY);
    70→    if (!raw) return [];
    71→    const parsed = JSON.parse(raw) as HistoryPoint[];
    72→    return Array.isArray(parsed) ? parsed.slice(-HISTORY_MAX) : [];
    73→  } catch {
    74→    return [];
    75→  }
    76→}

   303→          code: "client",
   304→          message,

   432→            "radial-gradient(700px 380px at 12% -5%, rgba(244,63,94,0.10), transparent 60%), radial-gradient(800px 420px at 88% 0%, rgba(245,158,11,0.07), transparent 60%)",
   433→        }}
