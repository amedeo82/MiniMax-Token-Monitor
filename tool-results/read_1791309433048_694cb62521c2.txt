    58→/** Scarica dal server lo storico completo e lo salva nel formato richiesto. */
    59→async function exportFullFromServer(format: "csv" | "json"): Promise<void> {
    60→  const server = await fetchServerHistory(10_000);
    61→  if (!server || server.points.length === 0) {
    62→    toast.error("Nessuno storico disponibile sul server");
    63→    return;
    64→  }
    65→  const points = serverPointsToHistory(server.points);
    66→  if (format === "csv") exportHistoryCsv(points);
    67→  else exportHistoryJson(points);
    68→  toast.success(`Esportati ${points.length} punti dallo storico server`);
    69→}

   372→interface TrendStats {
   373→  /** Velocità media di consumo in punti percento all'ora (positiva = consumo). */
   374→  ratePerHour: number | null;
   375→  /** Tempo stimato (ms) fino all'esaurimento, se il consumo prosegue lineare. */
   376→  exhaustionMs: number | null;
   377→  exhaustionAt: number | null;
   378→  samples: number;
   379→  windowMs: number;
   380→  /** I31: epoch ms dell'ultimo rilevamento usato per il trend. */
   381→  lastT: number | null;
   382→  /** I31: residuo osservato all'ultimo rilevamento (%). */
   383→  lastValue: number | null;
   384→}
   385→
   386→/** Statistiche derivate dalla storia: consumo medio e proiezione. */
   387→export function computeTrend(history: HistoryPoint[], win: "interval" | "weekly"): TrendStats {
   388→  const pts = history.slice(-60);
   389→  const key = win === "interval" ? "interval" : "weekly";
   390→  const last = pts.length > 0 ? pts[pts.length - 1] : null;
   391→  const base: TrendStats = {
   392→    ratePerHour: null,
   393→    exhaustionMs: null,
   394→    exhaustionAt: null,
   395→    samples: pts.length,
   396→    windowMs: last && pts.length > 1 ? last.t - pts[0].t : 0,
   397→    lastT: last ? last.t : null,
   398→    lastValue: last ? last[key] : null,
   399→  };
   400→  if (pts.length < 2 || !last) return base;
   401→
   402→  const first = pts[0];
   403→  const dtHours = (last.t - first.t) / 3600_000;
   404→  const dv = first[key] - last[key];
   405→  const ratePerHour = dtHours > 0 ? dv / dtHours : null;
   406→
   407→  if (ratePerHour !== null && ratePerHour > 0.01 && last[key] > 0) {
   408→    const hoursLeft = last[key] / ratePerHour;
   409→    return {
   410→      ...base,
   411→      ratePerHour,
   412→      exhaustionMs: hoursLeft * 3600_000,
   413→      exhaustionAt: last.t + hoursLeft * 3600_000,
   414→    };
   415→  }
   416→  return { ...base, ratePerHour };
   417→}
   418→
   419→/* ------------------------------------------------------------------ */
   420→/* I31 — Pace e quota sprecata (stesso filone del verdetto I23)        */
   421→/* ------------------------------------------------------------------ */
   422→
   423→/**
   424→ * Confronta il ritmo attuale con quello che servirebbe per sfruttare la
   425→ * finestra fino all'ultimo, quantificando:
   426→ * - il **ritmo di pareggio** (%/h che azzererebbe la quota ESATTAMENTE al reset);
   427→ * - la **quota sprecata**: residuo previsto al reset se il consumo proseguisse
   428→ *   al ritmo attuale (quota che il piano ricaricherà senza che sia stata usata);
   429→ * - il **fermo tecnico**: tempo senza quota tra l'esaurimento stimato e il
   430→ *   reset, quando il ritmo supera quello di pareggio.
   431→ *
   432→ * Tutto è calcolato sullo stesso modello lineare di I23 (ancorato all'ultimo
   433→ * rilevamento), così i numeri restano coerenti con il verdetto mostrato accanto.
   434→ */
   435→export interface PaceInfo {
   436→  /** Ore dall'ultimo rilevamento al reset; null se il reset è ignoto o già passato. */
   437→  hoursToReset: number | null;
   438→  /** Ritmo attuale (uguale a trend.ratePerHour), %/h. */
   439→  currentPace: number | null;
   440→  /** Ritmo di pareggio per azzerare la quota esattamente al reset, %/h. */
   441→  breakevenPace: number | null;
   442→  /** Residuo previsto al reset al ritmo attuale (punti %); 0 se si esaurisce prima. */
   443→  wastedQuota: number | null;
   444→  /** Tempo di consumo al ritmo attuale equivalente alla quota sprecata. */
   445→  wastedForMs: number | null;
   446→  /** Se la quota si esaurisce prima del reset: tempo «morto» fino al reset. */
   447→  downtimeMs: number | null;
   448→}
   449→
   450→const EMPTY_PACE: PaceInfo = {
   451→  hoursToReset: null,
   452→  currentPace: null,
   453→  breakevenPace: null,
   454→  wastedQuota: null,
   455→  wastedForMs: null,
   456→  downtimeMs: null,
   457→};
   458→
   459→export function computePace(trend: TrendStats, resetAt: number | null): PaceInfo {
   460→  if (resetAt === null || trend.lastT === null || trend.lastValue === null) {
   461→    return { ...EMPTY_PACE, currentPace: trend.ratePerHour };
   462→  }
   463→  const hoursToReset = (resetAt - trend.lastT) / 3600_000;
   464→  if (!isFinite(hoursToReset) || hoursToReset <= 0) {
   465→    return { ...EMPTY_PACE, currentPace: trend.ratePerHour };
   466→  }
   467→
   468→  const breakevenPace = trend.lastValue / hoursToReset;
   469→  const pace = trend.ratePerHour;
   470→
   471→  // Esaurimento stimato PRIMA del reset: niente quota sprecata, ma fermo tecnico
   472→  if (trend.exhaustionAt !== null && trend.exhaustionAt < resetAt) {
   473→    return {
   474→      hoursToReset,
   475→      currentPace: pace,
   476→      breakevenPace,
   477→      wastedQuota: 0,
   478→      wastedForMs: 0,
   479→      downtimeMs: Math.max(0, resetAt - trend.exhaustionAt),
   480→    };
   481→  }
   482→
   483→  // Residuo previsto al reset sul modello lineare (consumo ~null → residuo = attuale)
   484→  const residual = pace !== null ? trend.lastValue - pace * hoursToReset : trend.lastValue;
   485→  const wastedQuota = Math.max(0, residual);
   486→  return {
   487→    hoursToReset,
   488→    currentPace: pace,
   489→    breakevenPace,
   490→    wastedQuota,
   491→    wastedForMs: pace !== null && pace > 0.01 ? (wastedQuota / pace) * 3600_000 : null,
   492→    downtimeMs: null,
   493→  };
   494→}

   496→/** Pannello statistiche: consumo medio, proiezione esaurimento, rilevamenti, verdetto vs reset. */

   502→  serverCount = null,

   507→  resetAt?: number | null;

   509→  remaining?: number | null;
   510→  /** Punti conservati sul server (I16); null = server non raggiunto. */
   511→  serverCount?: number | null;

   528→            : `${trend.ratePerHour > 0 ? "+" : ""}${trend.ratePerHour.toFixed(2)} %/h`}

   575→/** Riga di verdetto: la quota durerà fino al reset o si esaurirà prima? */
   576→function ProjectionVerdict({
   577→  trend,
   578→  resetAt,
   579→  remaining,
   580→  win,
   581→}: {
   582→  trend: TrendStats;
   583→  resetAt: number | null;
   584→  remaining: number | null;
   585→  win: "interval" | "weekly";
   586→}) {
   587→  const winLabel = win === "interval" ? "di 5 ore" : "settimanale";
   588→  if (trend.samples < 2 || trend.ratePerHour === null) {
   589→    return (
   590→      <div className="rounded-xl border border-white/[0.07] bg-white/[0.02] p-4 sm:col-span-3">
   591→        <div className="flex items-start gap-2.5">
   592→          <Info className="mt-0.5 size-4 shrink-0 text-zinc-500" />
   593→          <div>
   594→            <p className="text-xs font-semibold text-zinc-300">Verdetto d’esaurimento</p>
   595→            <p className="mt-0.5 text-[11px] leading-relaxed text-zinc-500">
   596→              Raccogli almeno due rilevamenti con la finestra «{winLabel}» attiva per stimare se la quota
   597→              durerà fino al reset o si esaurirà prima.
   598→            </p>
   599→          </div>
   600→        </div>
   601→      </div>
   602→    );
   603→  }
   604→
   605→  const exhaustedBeforeReset =
   606→    trend.exhaustionAt !== null && resetAt !== null && trend.exhaustionAt < resetAt;
   607→
   608→  if (exhaustedBeforeReset) {
   609→    return (
   610→      <div className="rounded-xl border border-amber-400/25 bg-amber-400/[0.06] p-4 sm:col-span-3">
   611→        <div className="flex items-start gap-2.5">
   612→          <AlertTriangle className="mt-0.5 size-4 shrink-0 text-amber-400" />
   613→          <div>
   614→            <p className="text-xs font-semibold text-amber-200">Rischio esaurimento prima del reset</p>
   615→            <p className="mt-0.5 text-[11px] leading-relaxed text-amber-200/70">
   616→              Al ritmo attuale (+{trend.ratePerHour.toFixed(2)} %/h) la quota {winLabel} ({remaining?.toFixed(1)}%)
   617→              si esaurirà il <strong>{formatDateTimeRome(trend.exhaustionAt)}</strong>, PRIMA del reset previsto
   618→              per le <strong>{formatTimeRome(resetAt)}</strong>.
   619→            </p>
   620→          </div>
   621→        </div>
   622→      </div>
   623→    );
   624→  }
   625→
   626→  if (trend.exhaustionAt !== null && resetAt !== null) {
   627→    return (
   628→      <div className="rounded-xl border border-emerald-400/25 bg-emerald-400/[0.06] p-4 sm:col-span-3">
   629→        <div className="flex items-start gap-2.5">
   630→          <ShieldCheck className="mt-0.5 size-4 shrink-0 text-emerald-400" />
   631→          <div>
   632→            <p className="text-xs font-semibold text-emerald-200">Quota sufficiente fino al reset</p>
   633→            <p className="mt-0.5 text-[11px] leading-relaxed text-emerald-200/70">
   634→              Al ritmo attuale (+{trend.ratePerHour.toFixed(2)} %/h) l’esaurimento è previsto solo il{" "}
   635→              <strong>{formatDateTimeRome(trend.exhaustionAt)}</strong>: la finestra {winLabel} si ricaricherà
   636→              alle <strong>{formatTimeRome(resetAt)}</strong>, prima di esaurirsi.
   637→            </p>
   638→          </div>
   639→        </div>
   640→      </div>
   641→    );
   642→  }
   643→
   644→  // Consumo ~zero o negativo: nessun rischio stimato
   645→  return (
   646→    <div className="rounded-xl border border-white/[0.07] bg-white/[0.02] p-4 sm:col-span-3">
   647→      <div className="flex items-start gap-2.5">
   648→        <Info className="mt-0.5 size-4 shrink-0 text-zinc-500" />
   649→        <div>
   650→          <p className="text-xs font-semibold text-zinc-300">Nessun esaurimento previsto al ritmo attuale</p>
   651→          <p className="mt-0.5 text-[11px] leading-relaxed text-zinc-500">
   652→            {trend.ratePerHour > 0
   653→              ? `Il consumo è quasi nullo (+${trend.ratePerHour.toFixed(2)} %/h): la quota ${winLabel} durerà agevolmente fino al reset${resetAt ? ` delle ${formatTimeRome(resetAt)}` : ""}.`
   654→              : `Quota stabile o in recupero (reset avvenuto): nessun rischio di esaurimento per la finestra ${winLabel}.`}
   655→          </p>
   656→        </div>
   657→      </div>
   658→    </div>
   659→  );
   660→}
