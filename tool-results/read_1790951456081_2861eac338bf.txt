    56→    56→function loadSettings(): DashboardSettings {
    57→    57→  try {
    58→    58→    const raw = localStorage.getItem(SETTINGS_KEY);
    59→    59→    if (!raw) return DEFAULT_SETTINGS;
    60→    60→    const parsed = JSON.parse(raw) as Partial<DashboardSettings>;
    61→    61→    return { ...DEFAULT_SETTINGS, ...parsed };
    62→    62→  } catch {
    63→    63→    return DEFAULT_SETTINGS;
    64→    64→  }
    65→    65→}
    66→    66→
    67→    67→function loadHistory(): HistoryPoint[] {
    68→    68→  try {
    69→    69→    const raw = localStorage.getItem(HISTORY_KEY);
    70→    70→    if (!raw) return [];
    71→    71→    const parsed = JSON.parse(raw) as HistoryPoint[];
    72→    72→    return Array.isArray(parsed) ? parsed.slice(-HISTORY_MAX) : [];
    73→    73→  } catch {
    74→    74→    return [];
    75→    75→  }
    76→    76→}
