    51→const SETTINGS_KEY = "mm-dash-settings-v1";

    54→
    55→function loadSettings(): DashboardSettings {
    56→  try {
    57→    const raw = localStorage.getItem(SETTINGS_KEY);
    58→    if (!raw) return DEFAULT_SETTINGS;
    59→    const parsed = JSON.parse(raw) as Partial<DashboardSettings>;
    60→    return { ...DEFAULT_SETTINGS, ...parsed };
    61→  } catch {
    62→    return DEFAULT_SETTINGS;
    63→  }
    64→}
    65→
    66→function loadHistory(): HistoryPoint[] {
    67→  try {
    68→    const raw = localStorage.getItem(HISTORY_KEY);
    69→    if (!raw) return [];
    70→    const parsed = JSON.parse(raw) as HistoryPoint[];
    71→    return Array.isArray(parsed) ? parsed.slice(-HISTORY_MAX) : [];
    72→  } catch {
    73→    return [];
    74→  }
    75→}

   274→            setRawJson(json.data);
   275→            markOk();
   276→            pushHistory(snap);
   277→          } catch (e) {

   293→              code: "unknown",

   299→        const message = e instanceof Error ? e.message : "Errore imprevisto lato client.";

   301→        setError({
   302→          code: "client",
   303→          message,
   304→        });
   305→      } finally {
   306→        setLoading(false);
   307→      }
   308→    },
   309→    [pushHistory, markError, markOk],
   310→  );
   311→

   313→  React.useEffect(() => {

   348→    }

   354→      localStorage.setItem(SETTINGS_KEY, JSON.stringify(s));

   356→      /* noop */

   359→    setError(null);

   375→      setRawJson(raw);
   376→      setLastEndpoint("json://incollato (risposta API importata manualmente)");
   377→      markOk();
   378→      setLastFetch(snap.fetchedAt);
   379→      setError(null);
   380→      pushHistory(snap);
   381→      toast.success("JSON importato: analisi completata");
   382→    } catch (e) {
   383→      const message = e instanceof Error ? e.message : "JSON non interpretabile.";
   384→      setError({ code: "parse", message });
   385→      toast.error(message);
   386→    }
   387→  };
   388→
   389→  const handleClearEvents = () => {
   390→    clearEvents();
   391→    setEvents([]);
   392→    toast.success("Registro eventi svuotato");

   396→  const general = snapshot ? findModel(snapshot, "general") : undefined;
   397→  const otherModels = snapshot ? snapshot.models.filter((m) => m.modelName.toLowerCase() !== "general") : [];
   398→  const trend = React.useMemo(() => computeTrend(history, chartWindow), [history, chartWindow]);

   400→  const sourceLabel = !lastEndpoint
   401→    ? undefined
   402→    : lastEndpoint.startsWith("json://")
   403→      ? "Risposta JSON importata manualmente"
   404→      : lastEndpoint.startsWith("demo://")
   405→        ? "Modalità demo (nessuna chiamata reale)"
   406→        : lastEndpoint;

   421→

   426→      <div
   427→        aria-hidden
   428→        className="pointer-events-none fixed inset-0"
   429→        style={{
   430→          background:
   431→            "radial-gradient(700px 380px at 12% -5%, rgba(244,63,94,0.10), transparent 60%), radial-gradient(800px 420px at 88% 0%, rgba(245,158,11,0.07), transparent 60%)",
   432→        }}
   433→      />
   434→
   435→      <div className="relative flex min-h-screen flex-col">
   436→        {/* ------------------------------ Header ------------------------------ */}

   439→            <div className="flex items-center gap-3">
