    65→    65→function loadSettings(): DashboardSettings {
    66→    66→  try {
    67→    67→    const raw = localStorage.getItem(SETTINGS_KEY);
    68→    68→    if (!raw) return DEFAULT_SETTINGS;
    69→    69→    const parsed = JSON.parse(raw) as Partial<DashboardSettings> & Record<string, unknown>;
    70→    70→    /* I28 — migrazione delle vecchie soglie globali (alertThresholdWarn/Crit):
    71→    71→       i valori legacy si applicano a TUTTE le finestre, così il comportamento
    72→    72→       scelto dall'utente resta identico dopo l'aggiornamento. */
    73→    73→    const thresholds = migrateLegacyThresholds(parsed) ?? DEFAULT_SETTINGS.thresholds;
    74→    74→    const cleaned: Record<string, unknown> = { ...parsed };
    75→    75→    delete cleaned.alertThresholdWarn;
    76→    76→    delete cleaned.alertThresholdCrit;
    77→    77→    return { ...DEFAULT_SETTINGS, ...cleaned, thresholds } as DashboardSettings;
    78→    78→  } catch {
    79→    79→    return DEFAULT_SETTINGS;
    80→    80→  }
    81→    81→}

    83→    83→function loadHistory(): HistoryPoint[] {
    84→    84→  try {
    85→    85→    const raw = localStorage.getItem(HISTORY_KEY);
    86→    86→    if (!raw) return [];
    87→    87→    const parsed = JSON.parse(raw) as HistoryPoint[];
    88→    88→    return Array.isArray(parsed) ? parsed.slice(-HISTORY_MAX) : [];
    89→    89→  } catch {
    90→    90→    return [];
    91→    91→  }
    92→    92→}
